grammar MiniLang;

// PROGRAMA
// programa : PROGRAM ID COLON ---bloco DOT---- ;
// BLOCO E LINHAS
// bloco : BEGIN linhas END ; linhas : linha | linha linhas ; linha : comando SEMICOLON ;

// COMANDOS
// comando : atrib | io | controle | /* vazio */ ;
// ATRIBUIÇÃO
// atrib : ID ASSIGN expr ;

// ENTRADA E SAÍDA
// io : READ LPAREN ID RPAREN | WRITE LPAREN expr RPAREN ;

// ESTRUTURAS DE CONTROLE
// controle : condicional | loop ; condicional : IF relacional THEN bloco | IF relacional THEN bloco ELSE bloco ; loop : WHILE relacional DO bloco ;

// EXPRESSÕES ARITMÉTICAS
// expr : exprSimples | exprSimples PLUS expr | exprSimples MINUS expr ; exprSimples : termo | termo TIMES exprSimples | termo DIVIDE exprSimples ; termo : fator | LPAREN expr RPAREN ; fator : ID | NUMBER ;

// EXPRESSÕES RELACIONAIS
// relacional : expr relop expr ; relop : EQ | LT | GT ;

// TOKENS
// PROGRAM : 'program'; BEGIN : 'begin'; END : 'end'; READ : 'read'; WRITE : 'write'; IF : 'if'; THEN : 'then'; ELSE : 'else'; WHILE : 'while'; DO : 'do'; INT : 'int'; ASSIGN : '='; PLUS : '+'; MINUS : '-'; TIMES : '*'; DIVIDE : '/'; EQ : '=='; LT : '<'; GT : '>'; LPAREN : '('; RPAREN : ')'; LBRACE : '{'; RBRACE : '}'; SEMICOLON : ';'; COLON : ':'; COMMA : ','; DOT : '.'; ID : [a-z] [a-z0-9]*; NUMBER : [0-9]+; WHITESPACE : [ \t\r\n]+ -> skip; COMMENT : '//' ~[\r\n]* -> skip;
// programa : 'program' ':' 'nome'
